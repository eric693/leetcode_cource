#45 Jump Game II

Hard
Array Greedy Dynamic Programming

題目描述

給定一個非負整數陣列 nums，你最初位於陣列的第一個位置。

陣列中的每個元素代表你在該位置可以跳躍的最大長度。

你的目標是使用最少的跳躍次數到達陣列的最後一個位置。

假設你總是可以到達陣列的最後一個位置。

範例

範例 1：
輸入：nums = [2,3,1,1,4]
輸出：2
解釋：跳到最後一個位置的最小跳躍數是 2。從索引 0 跳 1 步到索引 1，然後跳 3 步到最後一個位置。
範例 2：
輸入：nums = [2,3,0,1,4]
輸出：2
範例 3：
輸入：nums = [1,2,3]
輸出：2
解釋：從索引 0 → 1 → 3

解題思路

核心概念：貪心算法（Greedy）

關鍵洞察：用最少的跳躍，每次跳盡可能遠。

  1. 維護當前跳躍能到達的最遠位置
  2. 遍歷當前跳躍範圍內的所有位置
  3. 找出下一跳能到達的最遠位置
  4. 當到達當前跳躍的邊界時，進行下一跳

詳細範例演示

完整跳躍過程：nums = [2,3,1,1,4]

       *   *───────→───→ ✓

貪心策略詳解

三個關鍵變數
jumps: 當前跳躍次數
current_end: 當前跳躍能到達的最遠邊界
farthest: 下一次跳躍能到達的最遠位置

策略:
1. 遍歷陣列（不包括最後一個元素）
2. 持續更新 farthest = max(farthest, i + nums[i])
3. 當 i 到達 current_end 時:
   - jumps += 1
   - current_end = farthest

為什麼不遍歷最後一個元素？
→ 因為到達最後一個元素前，我們已經計算出所需跳躍數

視覺化範例

跳躍範圍圖解

└─ 範圍 2: [索引 3-4] ← 從範圍 1 能到達

程式碼實現

# 方法一：貪心算法（最優解）
class Solution:
    def jump(self, nums: List[int]) -> int:
        n = len(nums)
        if n <= 1:
            return 0
        
        jumps = 0           # 跳躍次數
        current_end = 0     # 當前跳躍的邊界
        farthest = 0        # 能到達的最遠位置
        
        # 不需要遍歷最後一個元素
        for i in range(n - 1):
            # 更新能到達的最遠位置
            farthest = max(farthest, i + nums[i])
            
            # 到達當前跳躍的邊界
            if i == current_end:
                jumps += 1
                current_end = farthest
                
                # 優化：如果已經能到達終點，提前退出
                if current_end >= n - 1:
                    break
        
        return jumps


# 方法二：BFS 思想
class Solution:
    def jump(self, nums: List[int]) -> int:
        """
        把每次跳躍看作 BFS 的一層
        """
        n = len(nums)
        if n <= 1:
            return 0
        
        jumps = 0
        current_level_end = 0
        next_level_end = 0
        
        for i in range(n - 1):
            next_level_end = max(next_level_end, i + nums[i])
            
            if i == current_level_end:
                jumps += 1
                current_level_end = next_level_end
        
        return jumps


# 方法三：動態規劃（較慢，但易理解）
class Solution:
    def jump(self, nums: List[int]) -> int:
        """
        時間複雜度: O(n²)
        空間複雜度: O(n)
        """
        n = len(nums)
        dp = [float('inf')] * n
        dp[0] = 0
        
        for i in range(n):
            # 從位置 i 能跳到的所有位置
            for j in range(1, nums[i] + 1):
                if i + j < n:
                    dp[i + j] = min(dp[i + j], dp[i] + 1)
        
        return dp[n - 1]


# 方法四：詳細註解版本
class Solution:
    def jump(self, nums: List[int]) -> int:
        """
        貪心策略詳解版
        
        核心思想：
        每次跳躍時，我們在當前能到達的範圍內
        選擇能讓我們跳得最遠的位置
        """
        n = len(nums)
        
        # 邊界情況
        if n <= 1:
            return 0
        
        jumps = 0        # 跳躍次數
        current_end = 0  # 當前這一跳能到達的最遠邊界
        farthest = 0     # 在當前範圍內，下一跳能到達的最遠位置
        
        # 遍歷到倒數第二個位置即可
        # 因為最後一個位置是目標，不需要再跳
        for i in range(n - 1):
            # 更新在當前範圍內，下一跳能到達的最遠位置
            farthest = max(farthest, i + nums[i])
            
            # 到達當前這一跳的邊界了
            if i == current_end:
                # 必須進行下一跳
                jumps += 1
                # 更新邊界為下一跳能到達的最遠位置
                current_end = farthest
                
                # 提前退出優化
                if current_end >= n - 1:
                    break
        
        return jumps


# 方法五：反向貪心（從後往前）
class Solution:
    def jump(self, nums: List[int]) -> int:
        """
        從終點反向找，每次找能到達當前位置的最遠起點
        時間複雜度: O(n²)，較慢
        """
        n = len(nums)
        jumps = 0
        position = n - 1
        
        while position > 0:
            # 找最左邊能一步到達 position 的位置
            for i in range(position):
                if i + nums[i] >= position:
                    position = i
                    jumps += 1
                    break
        
        return jumps

// 方法一：貪心算法（最優解）
class Solution {
public:
    int jump(vector& nums) {
        int n = nums.size();
        if (n <= 1) return 0;
        
        int jumps = 0;
        int currentEnd = 0;
        int farthest = 0;
        
        for (int i = 0; i < n - 1; i++) {
            farthest = max(farthest, i + nums[i]);
            
            if (i == currentEnd) {
                jumps++;
                currentEnd = farthest;
                
                if (currentEnd >= n - 1) break;
            }
        }
        
        return jumps;
    }
};


// 方法二：動態規劃
class Solution {
public:
    int jump(vector& nums) {
        int n = nums.size();
        vector dp(n, INT_MAX);
        dp[0] = 0;
        
        for (int i = 0; i < n; i++) {
            for (int j = 1; j <= nums[i] && i + j < n; j++) {
                dp[i + j] = min(dp[i + j], dp[i] + 1);
            }
        }
        
        return dp[n - 1];
    }
};

class Solution {

        Arrays.fill(dp, Integer.MAX_VALUE);
        dp[0] = 0;

// 貪心（隱式 BFS）：在目前這一跳可到達的範圍內，找下一跳最遠能到哪
// 時間複雜度: O(n)，空間複雜度: O(1)
function jump(nums: number[]): number {
    let jumps = 0, curEnd = 0, farthest = 0;
    for (let i = 0; i < nums.length - 1; i++) {
        farthest = Math.max(farthest, i + nums[i]);
        if (i === curEnd) {        // 走到這一跳的邊界，必須再跳一次
            jumps++;
            curEnd = farthest;
        }
    }
    return jumps;
}
// 貪心（隱式 BFS）：在目前這一跳可到達的範圍內，找下一跳最遠能到哪
// 時間複雜度: O(n)，空間複雜度: O(1)
int jump(int* nums, int numsSize) {
    int jumps = 0, curEnd = 0, farthest = 0;
    for (int i = 0; i < numsSize - 1; i++) {
        if (i + nums[i] > farthest) farthest = i + nums[i];
        if (i == curEnd) {         // 走到這一跳的邊界，必須再跳一次
            jumps++;
            curEnd = farthest;
        }
    }
    return jumps;
}

複雜度分析

貪心算法：
  • 時間複雜度：O(n) - 只遍歷一次陣列
  • 空間複雜度：O(1) - 只用常數個變數
動態規劃：
  • 時間複雜度：O(n²) - 雙層循環
  • 空間複雜度：O(n) - DP 陣列

重點提示

關鍵技巧 1：不遍歷最後一個元素

range(n-1) 而非 range(n)，因為到達終點前已知答案。

關鍵技巧 2：邊界更新時機

當 i == current_end 時才跳躍，確保最少次數。

關鍵技巧 3：提前退出優化

如果 current_end >= n-1，可以提前結束。

常見錯誤：
  • 遍歷到最後一個元素（會多算一次跳躍）
  • 忘記更新 farthest（導致貪心策略失效）
  • 邊界條件處理不當（n=1 的情況）
  • 混淆 current_end 和 farthest 的作用
為什麼貪心策略是對的？
證明：
1. 每次跳躍時，我們選擇當前範圍內能到達最遠的位置
2. 這保證了下一次跳躍的選擇範圍最大
3. 因此能用最少的跳躍次數到達終點

反例思考：
如果不選擇能到達最遠的位置，
那麼下一次的選擇範圍會變小，
可能需要更多次跳躍。

因此貪心策略是最優的。

補充說明

BFS vs 貪心

兩種思路對比
BFS 思路：
把每次跳躍看作 BFS 的一層
第 k 層包含所有用 k 次跳躍能到達的位置

貪心思路：
直接計算每次跳躍的範圍
不需要顯式維護層級

兩者本質相同，但貪心實現更簡潔
