#215 Kth Largest Element in an Array

Medium
Heap Quickselect Sorting Divide and Conquer Array

題目描述

給定整數陣列 nums 和整數 k，請返回陣列中第 k 個最大的元素。

請注意，你需要找的是陣列排序後的第 k 個最大的元素，而不是第 k 個不同的元素。

你必須設計並實現時間複雜度為 O(n) 的演算法解決此問題。

範例

範例 1：
輸入：nums = [3,2,1,5,6,4], k = 2
輸出：5
解釋：排序後為 [1,2,3,4,5,6]，第 2 大是 5。
範例 2：
輸入：nums = [3,2,3,1,2,4,5,5,6], k = 4
輸出：4
解釋：重複元素各自計算，不去重。

解題思路

💡 三種解法的取捨
方法          時間           空間    適用情境
-----------------------------------------------------------------
排序後取      O(n log n)     O(1)    最簡單，資料小時夠用
最小堆積      O(n log k)     O(k)    ★ k 很小、資料串流、無法全載入記憶體
快速選擇      平均 O(n)      O(1)    ★ 符合題目 O(n) 要求，但最壞 O(n²)
              最壞 O(n²)

面試建議：先講排序法（30秒），再講堆積法，
          最後講快速選擇並說明它是題目要的 O(n)。

方法一：最小堆積（維護大小為 k 的堆）

🎯 堆頂永遠是「目前的第 k 大」
維護一個大小為 k 的【最小堆積】：
  堆中永遠保存「目前看過的最大的 k 個數」
  堆頂（最小的那個）就是第 k 大 ✓

for (num : nums) {
    heap.push(num);
    if (heap.size() > k) heap.pop();   // 彈出最小的
}
return heap.top();

【追蹤】nums = [3,2,1,5,6,4], k = 2
  push 3 → [3]
  push 2 → [2,3]
  push 1 → [1,2,3] 超過 k，pop 1 → [2,3]
  push 5 → [2,3,5] pop 2 → [3,5]
  push 6 → [3,5,6] pop 3 → [5,6]
  push 4 → [4,5,6] pop 4 → [5,6]
  堆頂 = 5 ✓

⭐ 為什麼是最小堆而不是最大堆？
   因為我們要不斷「淘汰最小的」，
   最小堆才能 O(1) 取得並 O(log k) 移除最小值。

方法二：快速選擇（Quickselect，本題正解）

⚡ 快速排序的「單邊」版本
第 k 大 = 升序排序後索引為 (n - k) 的元素

快速排序每次 partition 後，pivot 會落在它的最終位置。
如果那個位置正好是 n-k，就找到答案了！
否則只需要遞迴【其中一邊】（快排要遞迴兩邊）。

  if (pivotIndex == target) return nums[pivotIndex];
  if (pivotIndex < target)  只遞迴右邊;
  else                      只遞迴左邊;

【為什麼是 O(n)？】
每次期望排除一半：
  n + n/2 + n/4 + ... = 2n = O(n) ✓
（快排要處理兩邊，所以是 O(n log n)）

【最壞 O(n²)】
每次 pivot 都選到最大或最小值（如已排序陣列選第一個當 pivot）。
解法：【隨機選 pivot】，讓最壞情況的機率趨近於 0。

Partition 的實作（Lomuto 分割）

🔧 把小於 pivot 的都搬到左邊
int partition(vector& nums, int left, int right) {
    // 隨機選 pivot 並換到最右邊（避免最壞情況）
    int randIdx = left + rand() % (right - left + 1);
    swap(nums[randIdx], nums[right]);

    int pivot = nums[right];
    int i = left;                  // i 指向「下一個小於 pivot 的位置」

    for (int j = left; j < right; j++) {
        if (nums[j] < pivot) {
            swap(nums[i], nums[j]);
            i++;
        }
    }
    swap(nums[i], nums[right]);    // pivot 歸位
    return i;                      // 回傳 pivot 的最終索引
}

程式碼實現

# 方法一：最小堆積（O(n log k)，程式碼最短）
class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        import heapq
        heap = []
        for num in nums:
            heapq.heappush(heap, num)
            if len(heap) > k:
                heapq.heappop(heap)     # 彈出最小的
        return heap[0]                  # 堆頂 = 第 k 大


# 更簡潔：直接用 nlargest
class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        import heapq
        return heapq.nlargest(k, nums)[-1]


# 方法二：快速選擇（平均 O(n)，題目要求的解法）
class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        import random

        def partition(left, right):
            # 隨機選 pivot，避免最壞情況
            rand_idx = random.randint(left, right)
            nums[rand_idx], nums[right] = nums[right], nums[rand_idx]

            pivot = nums[right]
            i = left
            for j in range(left, right):
                if nums[j] < pivot:
                    nums[i], nums[j] = nums[j], nums[i]
                    i += 1
            nums[i], nums[right] = nums[right], nums[i]
            return i

        # 第 k 大 = 升序索引 n - k
        target = len(nums) - k
        left, right = 0, len(nums) - 1

        while True:
            p = partition(left, right)
            if p == target:
                return nums[p]
            elif p < target:
                left = p + 1          # 只往右邊找
            else:
                right = p - 1         # 只往左邊找
// 方法一：最小堆積（O(n log k)）
class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        // priority_queue 預設是最大堆，加 greater 變成最小堆
        priority_queue<int, vector<int>, greater<int>> minHeap;

        for (int num : nums) {
            minHeap.push(num);
            if (minHeap.size() > k) minHeap.pop();   // 彈出最小的
        }
        return minHeap.top();
    }
};

// 方法二：快速選擇（平均 O(n)，題目要求）
class Solution {
    int partition(vector<int>& nums, int left, int right) {
        // 隨機選 pivot，避免最壞情況
        int randIdx = left + rand() % (right - left + 1);
        swap(nums[randIdx], nums[right]);

        int pivot = nums[right];
        int i = left;
        for (int j = left; j < right; j++) {
            if (nums[j] < pivot) swap(nums[i++], nums[j]);
        }
        swap(nums[i], nums[right]);
        return i;
    }

public:
    int findKthLargest(vector<int>& nums, int k) {
        int target = nums.size() - k;    // 第 k 大 = 升序索引 n-k
        int left = 0, right = nums.size() - 1;

        while (true) {
            int p = partition(nums, left, right);
            if (p == target) return nums[p];
            else if (p < target) left = p + 1;    // 只找右邊
            else                 right = p - 1;   // 只找左邊
        }
    }
};

class Solution {
    public int findKthLargest(int[] nums, int k) {
        PriorityQueue<Integer> minHeap = new PriorityQueue<>();  // 預設最小堆

        for (int num : nums) {
            minHeap.offer(num);
            if (minHeap.size() > k) minHeap.poll();    // 彈出最小的
        }
        return minHeap.peek();
    }
}

// 方法二：快速選擇（平均 O(n)）
class Solution {
    private Random rand = new Random();

    public int findKthLargest(int[] nums, int k) {
        int target = nums.length - k;
        int left = 0, right = nums.length - 1;

        while (true) {
            int p = partition(nums, left, right);
            if (p == target) return nums[p];
            else if (p < target) left = p + 1;
            else                 right = p - 1;
        }
    }

    private int partition(int[] nums, int left, int right) {
        int randIdx = left + rand.nextInt(right - left + 1);
        swap(nums, randIdx, right);

        int pivot = nums[right], i = left;
        for (int j = left; j < right; j++) {
            if (nums[j] < pivot) swap(nums, i++, j);
        }
        swap(nums, i, right);
        return i;
    }

    private void swap(int[] a, int i, int j) {
        int t = a[i]; a[i] = a[j]; a[j] = t;
    }
}
// 快速選擇（三路切分，避免大量重複值退化）
// 平均時間複雜度: O(n)，空間複雜度: O(log n)
function findKthLargest(nums: number[], k: number): number {
    const select = (arr: number[], k: number): number => {
        const pivot = arr[Math.floor(Math.random() * arr.length)];
        const larger: number[] = [], equal: number[] = [], smaller: number[] = [];
        for (const x of arr) {
            if (x > pivot) larger.push(x);
            else if (x < pivot) smaller.push(x);
            else equal.push(x);
        }
        if (k <= larger.length) return select(larger, k);
        if (k <= larger.length + equal.length) return pivot;
        return select(smaller, k - larger.length - equal.length);
    };
    return select(nums, k);
}
// 最小堆：維持大小為 k 的堆，堆頂就是第 k 大
// 時間複雜度: O(n log k)，空間複雜度: O(k)
static void siftDown(int* h, int size, int i) {
    while (1) {
        int smallest = i, l = 2 * i + 1, r = 2 * i + 2;
        if (l < size && h[l] < h[smallest]) smallest = l;
        if (r < size && h[r] < h[smallest]) smallest = r;
        if (smallest == i) return;
        int tmp = h[i]; h[i] = h[smallest]; h[smallest] = tmp;
        i = smallest;
    }
}

int findKthLargest(int* nums, int numsSize, int k) {
    int* heap = malloc(sizeof(int) * k);
    for (int i = 0; i < k; i++) heap[i] = nums[i];
    for (int i = k / 2 - 1; i >= 0; i--) siftDown(heap, k, i);   // 建堆

    for (int i = k; i < numsSize; i++) {
        if (nums[i] > heap[0]) {         // 比堆頂大才換進來
            heap[0] = nums[i];
            siftDown(heap, k, 0);
        }
    }
    int ans = heap[0];
    free(heap);
    return ans;
}

複雜度分析

時間複雜度：
  • 排序法：O(n log n)
  • 最小堆積：O(n log k)，k 很小時接近 O(n)
  • 快速選擇：平均 O(n)，最壞 O(n²)（隨機化後機率極低）
  • 理論上有最壞 O(n) 的「中位數的中位數」演算法，但常數太大不實用
空間複雜度：
  • 最小堆積：O(k)
  • 快速選擇（迭代版）：O(1) ★
  • 快速選擇（遞迴版）：平均 O(log n) 堆疊

重點提示

💡 關鍵技巧 1：第 k 大 = 升序索引 n - k

轉換公式要背熟。也可以在 partition 時用降序排列，這樣目標索引就是 k-1。兩種寫法都可以，但務必在紙上驗證一次避免差一錯誤。

💡 關鍵技巧 2：隨機化 pivot 是必須的

LeetCode 的測試資料包含已排序的大陣列，如果固定選第一個或最後一個當 pivot，會退化成 O(n²) 而超時。加上隨機化只要一行，卻是能否通過的關鍵。

💡 關鍵技巧 3：Top-K 問題用「反向堆積」
  • 求最大的 k 個 → 用最小堆（淘汰最小的）
  • 求最小的 k 個 → 用最大堆（淘汰最大的）

這個「反直覺」的配對是 Top-K 問題的標準做法，一定要記住。

⚠️ 常見錯誤：
  • 索引轉換寫錯（n-k 還是 k-1）
  • 沒有隨機化 pivot，遇到已排序資料超時
  • 堆積用錯方向（求最大 k 個卻用最大堆）
  • 快速選擇遞迴了兩邊（那就變成快排 O(n log n) 了）
  • partition 的迴圈邊界寫成 j <= right（會把 pivot 自己也比較進去）

補充說明

Top-K 問題的四種解法

📊 依情境選擇
解法            時間          空間   適用情境
--------------------------------------------------------------------
全部排序        O(n log n)    O(1)   n 很小、或本來就要排序
大小 k 的堆積   O(n log k)    O(k)   ★ k << n、資料串流、分散式
快速選擇        平均 O(n)     O(1)   ★ 資料全在記憶體、只查一次
桶排序/計數     O(n)          O(範圍) 值域小且已知（如 #347）

【分散式場景】（面試系統設計常問）
  海量資料求 Top-K：
  1. 每台機器各自用堆積算出本地 Top-K
  2. 把所有本地結果匯總，再算一次全域 Top-K
  → 堆積法天然可以分治，快速選擇則不行

堆積（Heap）的核心操作複習

用陣列表示完全二元樹：
  節點 i 的左子 = 2i + 1，右子 = 2i + 2，父節點 = (i-1) / 2

操作        複雜度      說明
------------------------------------------------
取頂端      O(1)        arr[0]
插入        O(log n)    放到最後，向上調整（sift up）
刪除頂端    O(log n)    把最後一個移到頂，向下調整（sift down）
建堆        O(n)        ★ 從 n/2 開始逐個 sift down
                        （不是 O(n log n)！這是常見誤解）

各語言的堆積：
  C++    priority_queue（預設最大堆）
  Java   PriorityQueue（預設最小堆）★ 方向相反，要注意
  Python heapq（只有最小堆，要最大堆就存負數）

#295 資料串流中位數：雙堆積技巧

用兩個堆積把資料切成兩半：
  最大堆 small：存較小的一半（堆頂 = 較小半的最大值）
  最小堆 large：存較大的一半（堆頂 = 較大半的最小值）

保持 |size(small) - size(large)| <= 1

中位數：
  兩邊一樣大 → (small.top() + large.top()) / 2
  small 多一個 → small.top()

addNum: O(log n)，findMedian: O(1)

這是「用兩個堆積維護有序性」的經典技巧，
也是滑動視窗中位數（#480）的基礎。
