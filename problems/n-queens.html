#51 N-Queens

Hard
Backtracking Array Recursion

題目描述

n 皇后問題研究的是如何將 n 個皇后放置在 n×n 的棋盤上，並且使皇后彼此之間不能相互攻擊。

皇后的攻擊規則：

  • 同一行不能有兩個皇后
  • 同一列不能有兩個皇后
  • 同一對角線（正斜線和反斜線）不能有兩個皇后

給定整數 n，返回所有不同的 n 皇后問題的解決方案。

範例

範例 1：
輸入：n = 4

輸出：
[
 [".Q..",  // 解法 1
  "...Q",
  "Q...",
  "..Q."],

 ["..Q.",  // 解法 2
  "Q...",
  "...Q",
  ".Q.."]
]

解釋：4 皇后問題有兩個不同的解法。
範例 2：
輸入：n = 1
輸出：[["Q"]]

解題思路

💡 核心概念：回溯法（Backtracking）

策略：逐行放置皇后，每次檢查是否合法，不合法就回溯。

  1. 從第 0 行開始，依次嘗試在每一列放置皇后
  2. 檢查當前位置是否與已放置的皇后衝突
  3. 如果合法，放置皇后並遞迴處理下一行
  4. 如果所有行都放置完畢，記錄解答
  5. 回溯，嘗試其他位置

衝突檢測優化

🎯 使用集合快速檢測衝突

使用三個集合分別記錄：

  • cols：已佔用的列
  • diag1：已佔用的正斜線（row - col）
  • diag2：已佔用的反斜線（row + col）
對角線的數學規律：
- 正斜線（↗）：row - col 相同
  例如：(0,0), (1,1), (2,2) → row-col = 0
       (0,1), (1,2), (2,3) → row-col = -1

- 反斜線（↘）：row + col 相同
  例如：(0,3), (1,2), (2,1) → row+col = 3
       (0,2), (1,1), (2,0) → row+col = 2

# 方法一：使用集合優化（推薦）
class Solution:
    def solveNQueens(self, n: int) -> List[List[str]]:
        result = []
        board = [['.'] * n for _ in range(n)]
        
        # 使用集合記錄已佔用的列和對角線
        cols = set()        # 已佔用的列
        diag1 = set()       # 已佔用的正斜線 (row - col)
        diag2 = set()       # 已佔用的反斜線 (row + col)
        
        def backtrack(row):
            # 所有行都放置完畢
            if row == n:
                result.append([''.join(row) for row in board])
                return
            
            # 嘗試在當前行的每一列放置皇后
            for col in range(n):
                # 檢查是否衝突
                if (col in cols or 
                    (row - col) in diag1 or 
                    (row + col) in diag2):
                    continue
                
                # 放置皇后
                board[row][col] = 'Q'
                cols.add(col)
                diag1.add(row - col)
                diag2.add(row + col)
                
                # 遞迴處理下一行
                backtrack(row + 1)
                
                # 回溯，移除皇后
                board[row][col] = '.'
                cols.remove(col)
                diag1.remove(row - col)
                diag2.remove(row + col)
        
        backtrack(0)
        return result


# 方法二：使用陣列優化
class Solution:
    def solveNQueens(self, n: int) -> List[List[str]]:
        result = []
        queens = [-1] * n  # queens[i] 表示第 i 行皇后的列位置
        
        def is_valid(row, col):
            """檢查在 (row, col) 放置皇后是否合法"""
            for i in range(row):
                # 檢查列衝突
                if queens[i] == col:
                    return False
                # 檢查對角線衝突
                if abs(row - i) == abs(col - queens[i]):
                    return False
            return True
        
        def backtrack(row):
            if row == n:
                # 構建解答
                board = []
                for i in range(n):
                    row_str = '.' * queens[i] + 'Q' + '.' * (n - queens[i] - 1)
                    board.append(row_str)
                result.append(board)
                return
            
            for col in range(n):
                if is_valid(row, col):
                    queens[row] = col
                    backtrack(row + 1)
                    queens[row] = -1  # 回溯
        
        backtrack(0)
        return result


# 方法三：詳細註解版本
class Solution:
    def solveNQueens(self, n: int) -> List[List[str]]:
        """
        使用回溯法解決 N 皇后問題
        
        時間複雜度：O(N!)
        空間複雜度：O(N)
        """
        result = []
        board = [['.'] * n for _ in range(n)]
        
        # 優化：使用集合 O(1) 檢查衝突
        cols = set()    # 已佔用的列
        diag1 = set()   # 已佔用的主對角線
        diag2 = set()   # 已佔用的副對角線
        
        def backtrack(row):
            """
            在第 row 行放置皇后
            """
            # 基礎情況：所有行都處理完畢
            if row == n:
                # 將棋盤轉換為字串列表
                result.append([''.join(row) for row in board])
                return
            
            # 嘗試在當前行的每一列放置皇后
            for col in range(n):
                # 剪枝：檢查是否與已放置的皇后衝突
                if col in cols:
                    continue  # 列衝突
                if (row - col) in diag1:
                    continue  # 主對角線衝突
                if (row + col) in diag2:
                    continue  # 副對角線衝突
                
                # 做選擇：放置皇后
                board[row][col] = 'Q'
                cols.add(col)
                diag1.add(row - col)
                diag2.add(row + col)
                
                # 遞迴：處理下一行
                backtrack(row + 1)
                
                # 撤銷選擇：回溯
                board[row][col] = '.'
                cols.remove(col)
                diag1.remove(row - col)
                diag2.remove(row + col)
        
        # 從第 0 行開始
        backtrack(0)
        return result

class Solution {
public:
    vector> solveNQueens(int n) {
        vector> result;
        vector board(n, string(n, '.'));
        
        unordered_set cols;
        unordered_set diag1;
        unordered_set diag2;
        
        backtrack(0, n, board, result, cols, diag1, diag2);
        return result;
    }
    
private:
    void backtrack(int row, int n, vector& board,
                   vector>& result,
                   unordered_set& cols,
                   unordered_set& diag1,
                   unordered_set& diag2) {
        if (row == n) {
            result.push_back(board);
            return;
        }
        
        for (int col = 0; col < n; col++) {
            if (cols.count(col) || 
                diag1.count(row - col) || 
                diag2.count(row + col)) {
                continue;
            }
            
            board[row][col] = 'Q';
            cols.insert(col);
            diag1.insert(row - col);
            diag2.insert(row + col);
            
            backtrack(row + 1, n, board, result, cols, diag1, diag2);
            
            board[row][col] = '.';
            cols.erase(col);
            diag1.erase(row - col);
            diag2.erase(row + col);
        }
    }
};

class Solution {
    public List> solveNQueens(int n) {
        List> result = new ArrayList<>();
        char[][] board = new char[n][n];
        for (int i = 0; i < n; i++) {
            Arrays.fill(board[i], '.');
        }
        
        Set cols = new HashSet<>();
        Set diag1 = new HashSet<>();
        Set diag2 = new HashSet<>();
        
        backtrack(0, n, board, result, cols, diag1, diag2);
        return result;
    }
    
    private void backtrack(int row, int n, char[][] board,
                          List> result,
                          Set cols,
                          Set diag1,
                          Set diag2) {
        if (row == n) {
            result.add(construct(board));
            return;
        }
        
        for (int col = 0; col < n; col++) {
            if (cols.contains(col) || 
                diag1.contains(row - col) || 
                diag2.contains(row + col)) {
                continue;
            }
            
            board[row][col] = 'Q';
            cols.add(col);
            diag1.add(row - col);
            diag2.add(row + col);
            
            backtrack(row + 1, n, board, result, cols, diag1, diag2);
            
            board[row][col] = '.';
            cols.remove(col);
            diag1.remove(row - col);
            diag2.remove(row + col);
        }
    }
    
    private List construct(char[][] board) {
        List result = new ArrayList<>();
        for (char[] row : board) {
            result.add(new String(row));
        }
        return result;
    }
}

// 回溯：逐列放皇后，用三個集合記錄被佔用的欄與兩條對角線
// 時間複雜度: O(n!)，空間複雜度: O(n)
function solveNQueens(n: number): string[][] {
    const result: string[][] = [];
    const queens: number[] = [];                       // queens[row] = 該列皇后所在欄
    const cols = new Set<number>(), diag1 = new Set<number>(), diag2 = new Set<number>();

    const backtrack = (row: number): void => {
        if (row === n) {
            result.push(queens.map(c => '.'.repeat(c) + 'Q' + '.'.repeat(n - c - 1)));
            return;
        }
        for (let c = 0; c < n; c++) {
            if (cols.has(c) || diag1.has(row - c) || diag2.has(row + c)) continue;
            queens.push(c); cols.add(c); diag1.add(row - c); diag2.add(row + c);
            backtrack(row + 1);
            queens.pop(); cols.delete(c); diag1.delete(row - c); diag2.delete(row + c);
        }
    };

    backtrack(0);
    return result;
}
// 回溯：逐列放皇后，用三個布林陣列記錄被佔用的欄與兩條對角線（n ≤ 9）
// 時間複雜度: O(n!)，空間複雜度: O(n)
static char*** result;
static int count, cap, N;
static int queens[9];
static bool cols[9], diag1[17], diag2[17];

static void backtrack(int row) {
    if (row == N) {
        if (count == cap) {
            cap *= 2;
            result = realloc(result, sizeof(char**) * cap);
        }
        char** board = malloc(sizeof(char*) * N);
        for (int r = 0; r < N; r++) {
            board[r] = malloc(N + 1);
            memset(board[r], '.', N);
            board[r][queens[r]] = 'Q';
            board[r][N] = '\0';
        }
        result[count++] = board;
        return;
    }
    for (int c = 0; c < N; c++) {
        if (cols[c] || diag1[row - c + N - 1] || diag2[row + c]) continue;
        queens[row] = c;
        cols[c] = diag1[row - c + N - 1] = diag2[row + c] = true;
        backtrack(row + 1);
        cols[c] = diag1[row - c + N - 1] = diag2[row + c] = false;
    }
}

char*** solveNQueens(int n, int* returnSize, int** returnColumnSizes) {
    N = n;
    count = 0;
    cap = 16;
    result = malloc(sizeof(char**) * cap);
    backtrack(0);
    *returnSize = count;
    *returnColumnSizes = malloc(sizeof(int) * (count ? count : 1));
    for (int i = 0; i < count; i++) (*returnColumnSizes)[i] = n;
    return result;
}

複雜度分析

時間複雜度：O(N!)
  • 第一行有 N 種選擇
  • 第二行最多有 N-1 種選擇（排除衝突）
  • 依此類推，總共約 N! 種可能
  • 實際會因為剪枝而更少
空間複雜度：O(N)
  • 遞迴棧深度：O(N)
  • 集合空間：O(N)
  • 棋盤空間：O(N²)

重點提示

💡 關鍵技巧 1：對角線公式

正斜線 row-col 相同，反斜線 row+col 相同，記住這個規律！

💡 關鍵技巧 2：逐行放置

每行只放一個皇后，可以避免行衝突，簡化問題。

💡 關鍵技巧 3：集合優化

使用集合代替陣列檢查衝突，從 O(N) 優化到 O(1)。

⚠️ 常見錯誤：
  • 對角線公式記錯（row-col 和 row+col）
  • 回溯時忘記移除集合中的元素
  • 構建解答時沒有深拷貝棋盤
  • 邊界條件處理不當（n=1 的情況）
🎯 解的數量規律
n = 1 → 1 個解
n = 2 → 0 個解（無解）
n = 3 → 0 個解（無解）

def backtrack(路徑, 選擇列表):
    if 滿足結束條件:
        result.add(路徑)
        return
    
    for 選擇 in 選擇列表:
