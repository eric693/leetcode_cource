#44 Wildcard Matching

Hard
String Dynamic Programming Greedy

題目描述

給定一個字串 s 和一個字符模式 p，實現支援 '?' 和 '*' 萬用字符匹配的功能。

萬用字符規則：

  • '?'：匹配任意單個字符
  • '*'：匹配任意字符序列（包括空序列）

匹配應該覆蓋整個字串 s（不是部分字串）。

範例

範例 1：
輸入：s = "aa", p = "a"
輸出：false
解釋："a" 無法匹配 "aa" 整個字串
範例 2：
輸入：s = "aa", p = "*"
輸出：true
解釋：'*' 可以匹配任意字串
範例 3：
輸入：s = "cb", p = "?a"
輸出：false
解釋：'?' 匹配 'c'，但第二個字符是 'b' 而不是 'a'
範例 4：
輸入：s = "adceb", p = "*a*b"
輸出：true
解釋： 
第一個 '*' 匹配空字串
'a' 匹配 'a'
第二個 '*' 匹配 "dce"
'b' 匹配 'b'
範例 5：
輸入：s = "acdcb", p = "a*c?b"
輸出：false

解題思路

💡 核心概念：動態規劃 vs 貪心 + 回溯
這題有兩種主要解法：
  1. 動態規劃：O(m×n) 時間，O(m×n) 空間（可優化到 O(n)）
  2. 貪心 + 回溯：O(m+n) 最佳情況，O(m×n) 最壞情況，O(1) 空間

方法一：動態規劃（DP）

📊 DP 狀態定義

dp[i][j]：s 的前 i 個字符是否能被 p 的前 j 個字符匹配

狀態轉移：

1️⃣ 如果 p[j-1] == s[i-1] 或 p[j-1] == '?'
   dp[i][j] = dp[i-1][j-1]

2️⃣ 如果 p[j-1] == '*'
   dp[i][j] = dp[i][j-1]     // '*' 匹配空序列
           || dp[i-1][j]     // '*' 匹配一個或多個字符

3️⃣ 初始化：
   dp[0][0] = true           // 空字串匹配空模式
   dp[0][j] = dp[0][j-1] && p[j-1] == '*'  // 處理前導 '*'

DP 表格範例

s = "adceb", p = "*a*b"

dp ε * a * b
ε T T F F F
a F T T T F
d F T F T F
c F T F T F
e F T F T F
b F T F T T

方法二：貪心 + 回溯（雙指標）

🎯 貪心策略

核心思想：遇到 '*' 時，先假設它匹配空序列，如果後續匹配失敗，再回溯讓 '*' 多匹配一個字符。

使用四個指標：
- sIndex：當前處理 s 的位置
- pIndex：當前處理 p 的位置
- starIndex：最近一個 '*' 在 p 中的位置
- match：'*' 匹配到 s 的位置

匹配過程：
1. 如果字符匹配或 p[pIndex] == '?'：兩個指標都前進
2. 如果 p[pIndex] == '*'：記錄位置，p 前進，s 不動
3. 否則，如果之前遇到過 '*'：回溯，讓 '*' 多匹配一個字符
4. 否則：匹配失敗

程式碼實現

# 方法一：動態規劃（二維）
class Solution:
    def isMatch(self, s: str, p: str) -> bool:
        m, n = len(s), len(p)
        
        # dp[i][j]: s[:i] 是否匹配 p[:j]
        dp = [[False] * (n + 1) for _ in range(m + 1)]
        
        # 初始化
        dp[0][0] = True  # 空字串匹配空模式
        
        # 處理前導 '*'
        for j in range(1, n + 1):
            if p[j - 1] == '*':
                dp[0][j] = dp[0][j - 1]
        
        # 填表
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if p[j - 1] == '*':
                    # '*' 匹配空序列 或 匹配一個或多個字符
                    dp[i][j] = dp[i][j - 1] or dp[i - 1][j]
                elif p[j - 1] == '?' or s[i - 1] == p[j - 1]:
                    # 字符匹配
                    dp[i][j] = dp[i - 1][j - 1]
        
        return dp[m][n]


# 方法二：動態規劃（空間優化）
class Solution:
    def isMatch(self, s: str, p: str) -> bool:
        m, n = len(s), len(p)
        
        # 只需要兩行：當前行和上一行
        prev = [False] * (n + 1)
        prev[0] = True
        
        # 處理前導 '*'
        for j in range(1, n + 1):
            if p[j - 1] == '*':
                prev[j] = prev[j - 1]
        
        # 逐行更新
        for i in range(1, m + 1):
            curr = [False] * (n + 1)
            for j in range(1, n + 1):
                if p[j - 1] == '*':
                    curr[j] = curr[j - 1] or prev[j]
                elif p[j - 1] == '?' or s[i - 1] == p[j - 1]:
                    curr[j] = prev[j - 1]
            prev = curr
        
        return prev[n]


# 方法三：貪心 + 回溯（最優）
class Solution:
    def isMatch(self, s: str, p: str) -> bool:
        sIndex = pIndex = 0
        starIndex = match = -1
        
        while sIndex < len(s):
            # 情況 1：字符匹配或 '?'
            if pIndex < len(p) and (p[pIndex] == '?' or s[sIndex] == p[pIndex]):
                sIndex += 1
                pIndex += 1
            
            # 情況 2：遇到 '*'，記錄位置
            elif pIndex < len(p) and p[pIndex] == '*':
                starIndex = pIndex
                match = sIndex
                pIndex += 1  # '*' 先匹配空序列
            
            # 情況 3：回溯，讓 '*' 多匹配一個字符
            elif starIndex != -1:
                pIndex = starIndex + 1
                match += 1
                sIndex = match
            
            # 情況 4：匹配失敗
            else:
                return False
        
        # 處理 p 中剩餘的 '*'
        while pIndex < len(p) and p[pIndex] == '*':
            pIndex += 1
        
        return pIndex == len(p)


# 方法四：遞迴 + 記憶化
class Solution:
    def isMatch(self, s: str, p: str) -> bool:
        memo = {}
        
        def dp(i, j):
            # 已經計算過
            if (i, j) in memo:
                return memo[(i, j)]
            
            # 兩個都到達末尾
            if i == len(s) and j == len(p):
                return True
            
            # p 到達末尾，s 還有剩餘
            if j == len(p):
                return False
            
            # s 到達末尾，檢查 p 剩餘是否都是 '*'
            if i == len(s):
                result = all(c == '*' for c in p[j:])
                memo[(i, j)] = result
                return result
            
            # 字符匹配或 '?'
            if p[j] == '?' or s[i] == p[j]:
                result = dp(i + 1, j + 1)
            # '*'
            elif p[j] == '*':
                # '*' 匹配空序列 或 匹配一個或多個字符
                result = dp(i, j + 1) or dp(i + 1, j)
            else:
                result = False
            
            memo[(i, j)] = result
            return result
        
        return dp(0, 0)

                pIndex = starIndex + 1;
                match++;
                sIndex = match;
            }
            else {
                return false;

// 貪心 + 回溯：記住最後一個 '*' 的位置，失配時讓 '*' 多吃一個字元
// 時間複雜度: 平均 O(m + n)，最壞 O(m·n)；空間複雜度: O(1)
function isMatch(s: string, p: string): boolean {
    let i = 0, j = 0, star = -1, mark = 0;
    while (i < s.length) {
        if (j < p.length && (p[j] === '?' || p[j] === s[i])) {
            i++;
            j++;
        } else if (j < p.length && p[j] === '*') {
            star = j++;          // 先讓 '*' 匹配空字串
            mark = i;
        } else if (star !== -1) {
            j = star + 1;        // 回到 '*' 後面，讓 '*' 多吃一個字元
            i = ++mark;
        } else {
            return false;
        }
    }
    while (j < p.length && p[j] === '*') j++;
    return j === p.length;
}
// 貪心 + 回溯：記住最後一個 '*' 的位置，失配時讓 '*' 多吃一個字元
// 時間複雜度: 平均 O(m + n)，最壞 O(m·n)；空間複雜度: O(1)
bool isMatch(char* s, char* p) {
    int i = 0, j = 0, star = -1, mark = 0;
    while (s[i]) {
        if (p[j] && (p[j] == '?' || p[j] == s[i])) {
            i++;
            j++;
        } else if (p[j] == '*') {
            star = j++;          // 先讓 '*' 匹配空字串
            mark = i;
        } else if (star != -1) {
            j = star + 1;        // 回到 '*' 後面，讓 '*' 多吃一個字元
            i = ++mark;
        } else {
            return false;
        }
    }
    while (p[j] == '*') j++;
    return p[j] == '\0';
}

複雜度分析

動態規劃：
  • 時間複雜度：O(m × n)，m 和 n 分別是 s 和 p 的長度
  • 空間複雜度：O(m × n) 或 O(n)（優化後）
貪心 + 回溯：
  • 時間複雜度：O(m + n) 最佳情況，O(m × n) 最壞情況
  • 空間複雜度：O(1)
  • 實際執行通常比 DP 快很多

重點提示

💡 關鍵技巧 1：理解 '*' 的貪心策略

遇到 '*' 先假設匹配空序列，失敗時再回溯讓它多匹配字符。

💡 關鍵技巧 2：DP 狀態轉移

p[j] == '*' 時，dp[i][j] = dp[i][j-1] || dp[i-1][j]

💡 關鍵技巧 3：初始化處理

需要正確處理前導 '*' 的情況，它們可以匹配空字串。

⚠️ 常見錯誤：
  • 忘記處理 p 末尾的連續 '*'
  • DP 初始化錯誤，特別是前導 '*' 的情況
  • 貪心解法忘記保存 '*' 的位置用於回溯
  • 混淆 '?' 和 '*' 的匹配規則
🎯 貪心匹配詳細過程

步驟 2: p[1]='a' 匹配 s[0]='a'
        sIndex=1, pIndex=2

步驟 3: p[2]='*'，記錄 starIndex=2, match=1
        sIndex=1, pIndex=3

步驟 4: p[3]='b' 不匹配 s[1]='d'
        回溯到 starIndex=2, match=2
        sIndex=2, pIndex=3

步驟 5: p[3]='b' 不匹配 s[2]='c'

        sIndex=3, pIndex=3

步驟 6: p[3]='b' 不匹配 s[3]='e'
        繼續回溯, match=4
        sIndex=4, pIndex=3

步驟 7: p[3]='b' 匹配 s[4]='b'
        sIndex=5, pIndex=4

結束: sIndex=5=len(s), pIndex=4=len(p)
      返回 True
