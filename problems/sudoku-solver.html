#37 Sudoku Solver

Hard
Backtracking Array Hash Table Matrix

題目描述

編寫一個程序，通過填充空格來解決數獨問題。

數獨規則：

  • 數字 1-9 在每一行只能出現一次
  • 數字 1-9 在每一列只能出現一次
  • 數字 1-9 在每一個 3×3 宮格只能出現一次

說明：

  • 空白格用 '.' 表示
  • 給定的數獨序列只包含數字 1-9 和字符 '.'
  • 給定數獨永遠是 9×9 形式的
  • 給定的數獨只有唯一解

範例

範例：
輸入：
board = [
  ["5","3",".",".","7",".",".",".","."],
  ["6",".",".","1","9","5",".",".","."],
  [".","9","8",".",".",".",".","6","."],
  ["8",".",".",".","6",".",".",".","3"],
  ["4",".",".","8",".","3",".",".","1"],
  ["7",".",".",".","2",".",".",".","6"],
  [".","6",".",".",".",".","2","8","."],
  [".",".",".","4","1","9",".",".","5"],
  [".",".",".",".","8",".",".","7","9"]
]

輸出：
[
  ["5","3","4","6","7","8","9","1","2"],
  ["6","7","2","1","9","5","3","4","8"],
  ["1","9","8","3","4","2","5","6","7"],
  ["8","5","9","7","6","1","4","2","3"],
  ["4","2","6","8","5","3","7","9","1"],
  ["7","1","3","9","2","4","8","5","6"],
  ["9","6","1","5","3","7","2","8","4"],
  ["2","8","7","4","1","9","6","3","5"],
  ["3","4","5","2","8","6","1","7","9"]
]

解題思路

💡 核心概念：回溯法 + 剪枝

策略：逐個填入數字，遇到衝突就回溯。

  1. 找到下一個空格（'.'）
  2. 嘗試填入 1-9
  3. 檢查是否符合數獨規則
  4. 如果合法，遞迴填下一個空格
  5. 如果無法繼續，回溯並嘗試其他數字

優化技巧

🚀 三種優化策略
  1. 使用集合記錄：為每行、每列、每個宮格維護一個集合，O(1) 檢查衝突
  2. 位運算優化：使用位掩碼表示可用數字
  3. 選擇策略：優先填入候選數字最少的空格（更快剪枝）

3×3 宮格索引計算

📊 宮格編號規則
數獨 9×9 劃分為 9 個 3×3 宮格：

宮格編號：
┌─────┬─────┬─────┐
│  0  │  1  │  2  │
├─────┼─────┼─────┤
│  3  │  4  │  5  │
├─────┼─────┼─────┤
│  6  │  7  │  8  │
└─────┴─────┴─────┘

計算公式：
box_index = (row // 3) * 3 + (col // 3)

範例：
位置 (0, 0) → box 0
位置 (0, 5) → box 1
位置 (4, 4) → box 4
位置 (8, 8) → box 8

回溯過程示意

🔄 回溯搜尋樹
找到第一個空格 (0,2)
│
├─ 嘗試填 1 → 檢查規則 → 衝突 ✗
├─ 嘗試填 2 → 檢查規則 → 衝突 ✗
├─ 嘗試填 3 → 檢查規則 → 衝突 ✗
├─ 嘗試填 4 → 檢查規則 → 合法 ✓
│  │
│  └─ 遞迴填下一個空格
│     │
│     ├─ 嘗試各種數字...
│     └─ 如果無解 → 回溯到 (0,2)，嘗試 5
│
├─ 嘗試填 5 → ...
└─ 最終找到完整解答

# 方法一：回溯 + 集合優化（推薦）
class Solution:
    def solveSudoku(self, board: List[List[str]]) -> None:
        """
        Do not return anything, modify board in-place instead.
        """
        # 初始化集合，記錄每行、每列、每個宮格已使用的數字
        rows = [set() for _ in range(9)]
        cols = [set() for _ in range(9)]
        boxes = [set() for _ in range(9)]
        
        # 預處理：記錄已有數字
        for i in range(9):
            for j in range(9):
                if board[i][j] != '.':
                    num = board[i][j]
                    rows[i].add(num)
                    cols[j].add(num)
                    box_idx = (i // 3) * 3 + j // 3
                    boxes[box_idx].add(num)
        
        def backtrack(pos):
            # 基礎情況：已填完所有格子
            if pos == 81:
                return True
            
            row = pos // 9
            col = pos % 9
            
            # 跳過已填數字
            if board[row][col] != '.':
                return backtrack(pos + 1)
            
            box_idx = (row // 3) * 3 + col // 3
            
            # 嘗試填入 1-9
            for num in '123456789':
                # 檢查是否合法
                if (num in rows[row] or 
                    num in cols[col] or 
                    num in boxes[box_idx]):
                    continue
                
                # 做選擇
                board[row][col] = num
                rows[row].add(num)
                cols[col].add(num)
                boxes[box_idx].add(num)
                
                # 遞迴
                if backtrack(pos + 1):
                    return True
                
                # 撤銷選擇
                board[row][col] = '.'
                rows[row].remove(num)
                cols[col].remove(num)
                boxes[box_idx].remove(num)
            
            return False
        
        backtrack(0)


# 方法二：基礎回溯版本（易理解）
class Solution:
    def solveSudoku(self, board: List[List[str]]) -> None:
        def is_valid(row, col, num):
            """檢查在 (row, col) 填入 num 是否合法"""
            # 檢查行
            if num in board[row]:
                return False
            
            # 檢查列
            if num in [board[i][col] for i in range(9)]:
                return False
            
            # 檢查 3×3 宮格
            box_row, box_col = 3 * (row // 3), 3 * (col // 3)
            for i in range(box_row, box_row + 3):
                for j in range(box_col, box_col + 3):
                    if board[i][j] == num:
                        return False
            
            return True
        
        def backtrack():
            for i in range(9):
                for j in range(9):
                    if board[i][j] == '.':
                        for num in '123456789':
                            if is_valid(i, j, num):
                                board[i][j] = num
                                
                                if backtrack():
                                    return True
                                
                                board[i][j] = '.'
                        
                        return False
            
            return True
        
        backtrack()


# 方法三：位運算優化版本
class Solution:
    def solveSudoku(self, board: List[List[str]]) -> None:
        # 使用位掩碼表示可用數字（1-9）
        rows = [0] * 9
        cols = [0] * 9
        boxes = [0] * 9
        
        # 預處理
        for i in range(9):
            for j in range(9):
                if board[i][j] != '.':
                    digit = int(board[i][j]) - 1
                    box_idx = (i // 3) * 3 + j // 3
                    # 設置對應位為 1
                    rows[i] |= (1 << digit)
                    cols[j] |= (1 << digit)
                    boxes[box_idx] |= (1 << digit)
        
        def backtrack(pos):
            if pos == 81:
                return True
            
            row, col = pos // 9, pos % 9
            
            if board[row][col] != '.':
                return backtrack(pos + 1)
            
            box_idx = (row // 3) * 3 + col // 3
            # 計算可用數字（未被使用的位為 1）
            mask = ~(rows[row] | cols[col] | boxes[box_idx]) & 0x1ff
            
            while mask:
                # 獲取最低位的 1
                digit = (mask & -mask).bit_length() - 1
                num = str(digit + 1)
                
                # 做選擇
                board[row][col] = num
                rows[row] |= (1 << digit)
                cols[col] |= (1 << digit)
                boxes[box_idx] |= (1 << digit)
                
                if backtrack(pos + 1):
                    return True
                
                # 撤銷
                board[row][col] = '.'
                rows[row] ^= (1 << digit)
                cols[col] ^= (1 << digit)
                boxes[box_idx] ^= (1 << digit)
                
                # 清除最低位的 1
                mask &= (mask - 1)
            
            return False
        
        backtrack(0)

class Solution {
public:
    void solveSudoku(vector>& board) {
        vector> rows(9);
        vector> cols(9);
        vector> boxes(9);
        
        // 預處理
        for (int i = 0; i < 9; i++) {
            for (int j = 0; j < 9; j++) {
                if (board[i][j] != '.') {
                    char num = board[i][j];
                    rows[i].insert(num);
                    cols[j].insert(num);
                    int boxIdx = (i / 3) * 3 + j / 3;
                    boxes[boxIdx].insert(num);
                }
            }
        }
        
        backtrack(board, 0, rows, cols, boxes);
    }
    
private:
    bool backtrack(vector>& board, int pos,
                   vector>& rows,
                   vector>& cols,
                   vector>& boxes) {
        if (pos == 81) return true;
        
        int row = pos / 9;
        int col = pos % 9;
        
        if (board[row][col] != '.') {
            return backtrack(board, pos + 1, rows, cols, boxes);
        }
        
        int boxIdx = (row / 3) * 3 + col / 3;
        
        for (char num = '1'; num <= '9'; num++) {
            if (rows[row].count(num) || 
                cols[col].count(num) || 
                boxes[boxIdx].count(num)) {
                continue;
            }
            
            board[row][col] = num;
            rows[row].insert(num);
            cols[col].insert(num);
            boxes[boxIdx].insert(num);
            
            if (backtrack(board, pos + 1, rows, cols, boxes)) {
                return true;
            }
            
            board[row][col] = '.';
            rows[row].erase(num);
            cols[col].erase(num);
            boxes[boxIdx].erase(num);
        }
        
        return false;
    }
};

class Solution {
    public void solveSudoku(char[][] board) {
        Set[] rows = new HashSet[9];
        Set[] cols = new HashSet[9];
        Set[] boxes = new HashSet[9];
        
        for (int i = 0; i < 9; i++) {
            rows[i] = new HashSet<>();
            cols[i] = new HashSet<>();
            boxes[i] = new HashSet<>();
        }
        
        // 預處理
        for (int i = 0; i < 9; i++) {
            for (int j = 0; j < 9; j++) {
                if (board[i][j] != '.') {
                    char num = board[i][j];
                    rows[i].add(num);
                    cols[j].add(num);
                    int boxIdx = (i / 3) * 3 + j / 3;
                    boxes[boxIdx].add(num);
                }
            }
        }
        
        backtrack(board, 0, rows, cols, boxes);
    }
    
    private boolean backtrack(char[][] board, int pos,
                             Set[] rows,
                             Set[] cols,
                             Set[] boxes) {

        int col = pos % 9;

// 回溯 + 位元遮罩：rows / cols / boxes 各用 9 位元記錄已使用的數字
// 時間複雜度: 最壞 O(9^空格數)，實際因剪枝很快；空間複雜度: O(1)
function solveSudoku(board: string[][]): void {
    const rows = new Array(9).fill(0), cols = new Array(9).fill(0), boxes = new Array(9).fill(0);
    const empty: [number, number][] = [];
    const box = (i: number, j: number) => Math.floor(i / 3) * 3 + Math.floor(j / 3);

    for (let i = 0; i < 9; i++) {
        for (let j = 0; j < 9; j++) {
            if (board[i][j] === '.') {
                empty.push([i, j]);
            } else {
                const bit = 1 << (Number(board[i][j]) - 1);
                rows[i] |= bit; cols[j] |= bit; boxes[box(i, j)] |= bit;
            }
        }
    }

    const solve = (k: number): boolean => {
        if (k === empty.length) return true;
        const [i, j] = empty[k], b = box(i, j);
        for (let d = 0; d < 9; d++) {
            const bit = 1 << d;
            if ((rows[i] | cols[j] | boxes[b]) & bit) continue;   // 這個數字已被使用
            rows[i] |= bit; cols[j] |= bit; boxes[b] |= bit;
            board[i][j] = String(d + 1);
            if (solve(k + 1)) return true;
            rows[i] ^= bit; cols[j] ^= bit; boxes[b] ^= bit;      // 還原
        }
        board[i][j] = '.';
        return false;
    };

    solve(0);
}
// 回溯 + 位元遮罩：rows / cols / boxes 各用 9 位元記錄已使用的數字
// 時間複雜度: 最壞 O(9^空格數)，實際因剪枝很快；空間複雜度: O(1)
static int rows[9], cols[9], boxes[9];
static int empty[81][2], emptyCount;

static bool solve(char** board, int k) {
    if (k == emptyCount) return true;
    int i = empty[k][0], j = empty[k][1], b = (i / 3) * 3 + j / 3;
    for (int d = 0; d < 9; d++) {
        int bit = 1 << d;
        if ((rows[i] | cols[j] | boxes[b]) & bit) continue;   // 這個數字已被使用
        rows[i] |= bit; cols[j] |= bit; boxes[b] |= bit;
        board[i][j] = '1' + d;
        if (solve(board, k + 1)) return true;
        rows[i] ^= bit; cols[j] ^= bit; boxes[b] ^= bit;      // 還原
    }
    board[i][j] = '.';
    return false;
}

void solveSudoku(char** board, int boardSize, int* boardColSize) {
    memset(rows, 0, sizeof(rows));
    memset(cols, 0, sizeof(cols));
    memset(boxes, 0, sizeof(boxes));
    emptyCount = 0;
    for (int i = 0; i < 9; i++) {
        for (int j = 0; j < 9; j++) {
            if (board[i][j] == '.') {
                empty[emptyCount][0] = i;
                empty[emptyCount][1] = j;
                emptyCount++;
            } else {
                int bit = 1 << (board[i][j] - '1');
                rows[i] |= bit; cols[j] |= bit; boxes[(i / 3) * 3 + j / 3] |= bit;
            }
        }
    }
    solve(board, 0);
}

複雜度分析

時間複雜度：O(9^m)，m 是空格數量
  • 最壞情況下每個空格嘗試 9 個數字
  • 實際因剪枝會快很多
空間複雜度：O(1)
  • 遞迴深度最多 81（9×9）
  • 集合空間固定（9 行 + 9 列 + 9 宮格）

重點提示

💡 關鍵技巧 1：宮格索引公式

box_index = (row // 3) * 3 + (col // 3)，必須記住！

💡 關鍵技巧 2：集合優化

使用集合記錄已用數字，檢查從 O(9) 優化到 O(1)。

💡 關鍵技巧 3：預處理

開始前先將已有數字加入集合，避免重複檢查。

⚠️ 常見錯誤：
  • 宮格索引計算錯誤
  • 回溯時忘記從集合中移除數字
  • 遍歷方式低效（應該用位置索引而非雙層循環）
  • 沒有檢查題目是否有解（本題保證有唯一解）
