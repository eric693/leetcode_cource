#210 Course Schedule II

Medium
Graph Topological Sort BFS DFS

題目描述

現在你總共有 numCourses 門課需要選，記為 0 到 numCourses - 1。給你一個陣列 prerequisites，其中 prerequisites[i] = [ai, bi] 表示在選修課程 ai 前必須先選修 bi。

返回你為了學完所有課程所安排的學習順序。可能會有多個正確的順序，你只要返回任意一種就可以了。如果不可能完成所有課程，返回空陣列。

這題就是 #207 的進階版：不只判斷有沒有環，還要輸出具體的拓撲序。

範例

範例 1：
輸入：numCourses = 2, prerequisites = [[1,0]]
輸出：[0,1]
範例 2：
輸入：numCourses = 4, prerequisites = [[1,0],[2,0],[3,1],[3,2]]
輸出：[0,1,2,3] 或 [0,2,1,3]
解釋：課程 0 是課程 1 和 2 的先修課，而 1 和 2 都是 3 的先修課。
範例 3：
輸入：numCourses = 1, prerequisites = []
輸出：[0]

解題思路

💡 BFS 拓撲排序（Kahn 演算法）
和 #207 完全一樣，只是把「出隊的節點」依序記錄下來。

【演算法】
1. 建圖（b → a）並統計每個節點的入度
2. 所有入度為 0 的節點入隊
3. 每次出隊一個節點 u：
   - 把 u 加入結果
   - 對每個後繼 v：indegree[v]--，若變成 0 就入隊
4. 若結果長度 == numCourses → 返回結果
   否則有環 → 返回空陣列

【完整追蹤】
numCourses = 4, prerequisites = [[1,0],[2,0],[3,1],[3,2]]

圖：0→1, 0→2, 1→3, 2→3
入度：[0, 1, 1, 2]

佇列 = [0]，order = []
出隊 0 → order = [0]
  1 的入度 1→0，入隊
  2 的入度 1→0，入隊
佇列 = [1, 2]
出隊 1 → order = [0, 1]
  3 的入度 2→1
出隊 2 → order = [0, 1, 2]
  3 的入度 1→0，入隊
出隊 3 → order = [0, 1, 2, 3]

長度 4 == numCourses → 返回 [0,1,2,3] ✓

方法二：DFS 後序 + 反轉

🔄 「完成時間」的逆序就是拓撲序
DFS 的核心洞察：
  一個節點「完成」（所有後繼都走完）時，
  它必定排在所有後繼的【前面】。

所以把節點依「完成順序」記錄下來，
最後【反轉】就是拓撲序。

0→1, 0→2, 1→3, 2→3

dfs(0)
  dfs(1)
    dfs(3) → 3 沒有後繼，完成 → post = [3]
    1 完成 → post = [3, 1]
  dfs(2)
    3 已完成（黑色），跳過
    2 完成 → post = [3, 1, 2]
  0 完成 → post = [3, 1, 2, 0]

反轉 → [0, 2, 1, 3] ✓（也是合法的拓撲序）

同時用三色標記偵測環：遇到灰色節點就返回空陣列。

拓撲序不唯一

當多個節點同時入度為 0 時，任何順序都合法。所以本題允許返回任意一種答案。若要求字典序最小的拓撲序，把佇列換成最小堆積（priority queue）即可。

程式碼實現

# 方法一：BFS 拓撲排序（Kahn，推薦）
class Solution:
    def findOrder(self, numCourses: int, prerequisites: List[List[int]]) -> List[int]:
        from collections import deque

        graph = [[] for _ in range(numCourses)]
        indegree = [0] * numCourses

        for course, pre in prerequisites:
            graph[pre].append(course)      # pre → course
            indegree[course] += 1

        queue = deque(i for i in range(numCourses) if indegree[i] == 0)
        order = []

        while queue:
            cur = queue.popleft()
            order.append(cur)              # 記錄出隊順序
            for nxt in graph[cur]:
                indegree[nxt] -= 1
                if indegree[nxt] == 0:
                    queue.append(nxt)

        # 若無法排出全部課程，代表有環
        return order if len(order) == numCourses else []


# 方法二：DFS 後序 + 反轉
class Solution:
    def findOrder(self, numCourses: int, prerequisites: List[List[int]]) -> List[int]:
        graph = [[] for _ in range(numCourses)]
        for course, pre in prerequisites:
            graph[pre].append(course)

        state = [0] * numCourses   # 0=未訪問, 1=遞迴中, 2=已完成
        post = []

        def dfs(node):
            if state[node] == 1:
                return False        # 遇到灰色 → 有環
            if state[node] == 2:
                return True

            state[node] = 1
            for nxt in graph[node]:
                if not dfs(nxt):
                    return False
            state[node] = 2
            post.append(node)       # 完成時記錄
            return True

        for i in range(numCourses):
            if not dfs(i):
                return []

        return post[::-1]           # 反轉完成順序
// 方法一：BFS 拓撲排序（Kahn，推薦）
class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>> graph(numCourses);
        vector<int> indegree(numCourses, 0);

        for (auto& p : prerequisites) {
            graph[p[1]].push_back(p[0]);      // pre → course
            indegree[p[0]]++;
        }

        queue<int> q;
        for (int i = 0; i < numCourses; i++) {
            if (indegree[i] == 0) q.push(i);
        }

        vector<int> order;
        while (!q.empty()) {
            int cur = q.front(); q.pop();
            order.push_back(cur);             // 記錄出隊順序
            for (int nxt : graph[cur]) {
                if (--indegree[nxt] == 0) q.push(nxt);
            }
        }

        if (order.size() != numCourses) return {};   // 有環
        return order;
    }
};

// 方法一：BFS 拓撲排序（Kahn，推薦）
class Solution {
    public int[] findOrder(int numCourses, int[][] prerequisites) {
        List<List<Integer>> graph = new ArrayList<>();
        for (int i = 0; i < numCourses; i++) graph.add(new ArrayList<>());
        int[] indegree = new int[numCourses];

        for (int[] p : prerequisites) {
            graph.get(p[1]).add(p[0]);        // pre → course
            indegree[p[0]]++;
        }

        Deque<Integer> queue = new ArrayDeque<>();
        for (int i = 0; i < numCourses; i++) {
            if (indegree[i] == 0) queue.offer(i);
        }

        int[] order = new int[numCourses];
        int idx = 0;
        while (!queue.isEmpty()) {
            int cur = queue.poll();
            order[idx++] = cur;               // 記錄出隊順序
            for (int nxt : graph.get(cur)) {
                if (--indegree[nxt] == 0) queue.offer(nxt);
            }
        }

        return idx == numCourses ? order : new int[0];   // 有環回傳空陣列
    }
}
// Kahn 拓撲排序：不斷取出入度為 0 的課程
// 時間複雜度: O(V + E)，空間複雜度: O(V + E)
function findOrder(numCourses: number, prerequisites: number[][]): number[] {
    const graph: number[][] = Array.from({ length: numCourses }, () => []);
    const indeg = new Array(numCourses).fill(0);
    for (const [course, pre] of prerequisites) {
        graph[pre].push(course);
        indeg[course]++;
    }

    const order: number[] = [];
    for (let i = 0; i < numCourses; i++) if (indeg[i] === 0) order.push(i);

    // order 本身就當作佇列使用
    for (let head = 0; head < order.length; head++) {
        for (const next of graph[order[head]]) {
            if (--indeg[next] === 0) order.push(next);
        }
    }
    return order.length === numCourses ? order : [];
}
// Kahn 拓撲排序：不斷取出入度為 0 的課程
// 時間複雜度: O(V + E)，空間複雜度: O(V + E)
int* findOrder(int numCourses, int** prerequisites, int prerequisitesSize,
               int* prerequisitesColSize, int* returnSize) {
    // 用「前向星」存圖：head[u] 指向 u 的第一條邊
    int* head = malloc(sizeof(int) * numCourses);
    int* nxt = malloc(sizeof(int) * (prerequisitesSize + 1));
    int* to = malloc(sizeof(int) * (prerequisitesSize + 1));
    int* indeg = calloc(numCourses, sizeof(int));
    for (int i = 0; i < numCourses; i++) head[i] = -1;

    for (int e = 0; e < prerequisitesSize; e++) {
        int course = prerequisites[e][0], pre = prerequisites[e][1];
        to[e] = course;
        nxt[e] = head[pre];
        head[pre] = e;
        indeg[course]++;
    }

    int* order = malloc(sizeof(int) * numCourses);
    int tail = 0;
    for (int i = 0; i < numCourses; i++) if (indeg[i] == 0) order[tail++] = i;
    for (int h = 0; h < tail; h++) {
        for (int e = head[order[h]]; e != -1; e = nxt[e]) {
            if (--indeg[to[e]] == 0) order[tail++] = to[e];
        }
    }

    free(head); free(nxt); free(to); free(indeg);
    *returnSize = (tail == numCourses) ? numCourses : 0;   // 有環則回傳空陣列
    return order;
}

複雜度分析

時間複雜度：O(V + E)
  • V = numCourses，E = prerequisites 長度
  • 若要求字典序最小（用堆積）則為 O(V log V + E)
空間複雜度：O(V + E)，鄰接表 + 入度陣列 + 佇列

重點提示

💡 關鍵技巧 1：Kahn 演算法只比 #207 多一行

order.push_back(cur)。所以這兩題應該一起準備，掌握一個就等於掌握兩個。

💡 關鍵技巧 2：DFS 版要記得反轉

DFS 記錄的是「完成順序」，而拓撲序是完成順序的逆序。忘記反轉是最常見的錯誤。

💡 面試延伸：字典序最小的拓撲序

把 Kahn 演算法中的普通佇列換成最小堆積，每次都優先取出編號最小的可用節點。這是很常見的追問，答案非常簡單但要想得到。

⚠️ 常見錯誤：
  • 建圖方向反了（[a,b] 是 b → a）
  • 忘記檢查結果長度，有環時仍返回不完整的順序
  • DFS 版忘記反轉結果
  • 誤以為拓撲序唯一而執著於配對某個特定答案

補充說明

拓撲排序的三個必記結論

📝 面試觀念題
1【存在性】一個有向圖存在拓撲序 ⟺ 它是 DAG（無環有向圖）

2【唯一性】拓撲序唯一 ⟺ 每一步都恰好只有一個入度為 0 的節點
           （等價於：存在一條經過所有節點的哈密頓路徑）

3【應用】DAG 上的動態規劃必須按拓撲序進行，
         因為要保證計算某個節點時，它的所有前驅都已算完。

DAG 上的 DP：最長路徑

一般圖的最長路徑是 NP-Hard，
但【DAG 上的最長路徑】可以在 O(V+E) 解決：

按拓撲序處理每個節點 u：
    for (v : graph[u])
        dp[v] = max(dp[v], dp[u] + 1);

因為拓撲序保證處理 u 時，
所有能到達 u 的節點都已經計算完畢。

應用：#329 矩陣中的最長遞增路徑
      （把「較小值 → 較大值」視為有向邊，天然是 DAG）

字典序最小的拓撲序實作

// 只要把 queue 換成 min-heap
priority_queue<int, vector<int>, greater<int>> pq;
for (int i = 0; i < numCourses; i++)
    if (indegree[i] == 0) pq.push(i);

vector<int> order;
while (!pq.empty()) {
    int cur = pq.top(); pq.pop();     // 每次取編號最小的
    order.push_back(cur);
    for (int nxt : graph[cur])
        if (--indegree[nxt] == 0) pq.push(nxt);
}

複雜度變成 O(V log V + E)。
這在「任務排程需要穩定輸出」的實務場景很常用。
