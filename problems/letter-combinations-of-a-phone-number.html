#17 Letter Combinations of a Phone Number

Medium
Backtracking String Hash Table

題目描述

給定一個僅包含數字 2-9 的字串，返回所有它能表示的字母組合。答案可以按任意順序返回。

數字到字母的映射與電話按鍵相同（注意 1 不對應任何字母）：

2 → abc      3 → def      4 → ghi
5 → jkl      6 → mno      7 → pqrs
8 → tuv      9 → wxyz

範例

範例 1：
輸入：digits = "23"
輸出：["ad","ae","af","bd","be","bf","cd","ce","cf"]
解釋：3 × 3 = 9 種組合。
範例 2：
輸入：digits = ""
輸出：[]
解釋：空輸入要返回空陣列，不是 [""]。
範例 3：
輸入：digits = "2"
輸出：["a","b","c"]

解題思路

💡 核心：多集合的笛卡爾積 = 多層回溯
digits = "23"
  第 1 層：從 {a, b, c} 選一個
  第 2 層：從 {d, e, f} 選一個

                ""
        /       |       \
      "a"      "b"      "c"
     / | \    / | \    / | \
   ad ae af bd be bf cd ce cf

深度 = digits 的長度
每層的分支數 = 該數字對應的字母數（3 或 4）

⭐ 與其他回溯題的差別：
   每一層的「選擇列表」是不同的（由該位數字決定），
   而不是共用同一個候選集合。

回溯實作

index 表示「現在處理第幾個數字」
void backtrack(int index) {
    if (index == digits.size()) {      // 所有數字都處理完
        result.push_back(path);
        return;
    }

    string letters = phone[digits[index]];   // 這一位對應的字母集
    for (char c : letters) {
        path.push_back(c);              // ① 做選擇
        backtrack(index + 1);           // ② 處理下一位數字
        path.pop_back();                // ③ 撤銷選擇
    }
}

沒有剪枝條件 —— 因為所有組合都是合法答案。

方法二：BFS 迭代（不用遞迴）

🔄 逐位擴展佇列
result = [""]

處理 '2'：把 result 中每個字串各接上 a、b、c
  [""] → ["a", "b", "c"]

處理 '3'：把 result 中每個字串各接上 d、e、f
  ["a","b","c"] → ["ad","ae","af","bd","be","bf","cd","ce","cf"] ✓

每一輪答案數量乘以該數字的字母數。
優點：沒有遞迴堆疊，程式碼也很短。

程式碼實現

# 方法一：回溯（推薦）
class Solution:
    def letterCombinations(self, digits: str) -> List[str]:
        if not digits:
            return []                 # 注意：空輸入回傳 []，不是 [""]

        phone = {
            '2': 'abc', '3': 'def', '4': 'ghi', '5': 'jkl',
            '6': 'mno', '7': 'pqrs', '8': 'tuv', '9': 'wxyz'
        }

        result = []
        path = []

        def backtrack(index):
            # 所有數字都處理完了
            if index == len(digits):
                result.append(''.join(path))
                return

            for ch in phone[digits[index]]:
                path.append(ch)          # ① 做選擇
                backtrack(index + 1)     # ② 下一位數字
                path.pop()               # ③ 撤銷選擇

        backtrack(0)
        return result

class Solution:
    def letterCombinations(self, digits: str) -> List[str]:
        if not digits:
            return []

        phone = {
            '2': 'abc', '3': 'def', '4': 'ghi', '5': 'jkl',
            '6': 'mno', '7': 'pqrs', '8': 'tuv', '9': 'wxyz'
        }

        result = ['']
        for d in digits:
            # 每一輪把現有結果與新字母做笛卡爾積
            result = [prefix + ch for prefix in result for ch in phone[d]]
        return result
// 方法一：回溯（推薦）
class Solution {
    vector<string> result;
    string path;
    vector<string> phone = {
        "", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"
    };

    void backtrack(const string& digits, int index) {
        if (index == digits.size()) {
            result.push_back(path);
            return;
        }

        const string& letters = phone[digits[index] - '0'];
        for (char c : letters) {
            path.push_back(c);                  // ① 做選擇
            backtrack(digits, index + 1);       // ② 下一位
            path.pop_back();                    // ③ 撤銷
        }
    }

public:
    vector<string> letterCombinations(string digits) {
        if (digits.empty()) return {};          // 空輸入回傳空陣列
        backtrack(digits, 0);
        return result;
    }
};

// 方法二：BFS 迭代
class Solution {
public:
    vector<string> letterCombinations(string digits) {
        if (digits.empty()) return {};
        vector<string> phone = {"","","abc","def","ghi","jkl",
                                "mno","pqrs","tuv","wxyz"};
        vector<string> result = {""};
        for (char d : digits) {
            vector<string> next;
            for (const string& prefix : result)
                for (char c : phone[d - '0'])
                    next.push_back(prefix + c);
            result = move(next);
        }
        return result;
    }
};
// 方法一：回溯（推薦）
class Solution {
    private List<String> result = new ArrayList<>();
    private StringBuilder path = new StringBuilder();
    private String[] phone = {
        "", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"
    };

    public List<String> letterCombinations(String digits) {
        if (digits == null || digits.isEmpty()) return result;
        backtrack(digits, 0);
        return result;
    }

    private void backtrack(String digits, int index) {
        if (index == digits.length()) {
            result.add(path.toString());
            return;
        }

        String letters = phone[digits.charAt(index) - '0'];
        for (char c : letters.toCharArray()) {
            path.append(c);                          // ① 做選擇
            backtrack(digits, index + 1);            // ② 下一位
            path.deleteCharAt(path.length() - 1);    // ③ 撤銷
        }
    }
}
// 回溯：每一位數字選一個對應字母
// 時間複雜度: O(4^n · n)，空間複雜度: O(n)
function letterCombinations(digits: string): string[] {
    if (!digits) return [];
    const map = ['', '', 'abc', 'def', 'ghi', 'jkl', 'mno', 'pqrs', 'tuv', 'wxyz'];
    const result: string[] = [];

    const backtrack = (index: number, path: string): void => {
        if (index === digits.length) {
            result.push(path);
            return;
        }
        for (const ch of map[Number(digits[index])]) {
            backtrack(index + 1, path + ch);
        }
    };

    backtrack(0, '');
    return result;
}
// 回溯：每一位數字選一個對應字母（最多 4^4 = 256 種組合）
// 時間複雜度: O(4^n · n)，空間複雜度: O(n)
static const char* MAP[10] = {"", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};
static char** result;
static int count;
static char path[5];

static void backtrack(char* digits, int index, int n) {
    if (index == n) {
        path[n] = '\0';
        result[count++] = strdup(path);
        return;
    }
    for (const char* p = MAP[digits[index] - '0']; *p; p++) {
        path[index] = *p;
        backtrack(digits, index + 1, n);
    }
}

char** letterCombinations(char* digits, int* returnSize) {
    int n = strlen(digits);
    count = 0;
    result = malloc(sizeof(char*) * 256);
    if (n > 0) backtrack(digits, 0, n);
    *returnSize = count;
    return result;
}

複雜度分析

時間複雜度：O(4ⁿ × n)
  • n 為 digits 長度，每位最多 4 個字母（7 和 9）
  • 共最多 4ⁿ 個組合，每個組合建構需 O(n)
  • 這是輸出規模的下界
空間複雜度：O(n)
  • 遞迴堆疊深度 n，path 長度 n
  • 不計輸出所需的 O(4ⁿ × n)

重點提示

💡 關鍵技巧 1：空輸入要回傳 []，不是 [""]

這是本題最常見的失分點。如果不特判空字串，回溯會直接觸發終止條件而把空字串加入結果，得到 [""]，與題目要求不符。

💡 關鍵技巧 2：每層的選擇列表不同

本題與 #46、#78 的結構差異在於：選擇列表由當前層決定（phone[digits[index]]），而不是固定的 nums。理解這點，就能處理各種「多維笛卡爾積」型的題目。

💡 用陣列取代 HashMap

因為 key 是 '2'~'9'，直接用長度 10 的字串陣列、以 digits[i] - '0' 當索引，比 HashMap 更快也更簡潔。這是「key 是小範圍整數時用陣列」的通用優化。

⚠️ 常見錯誤：
  • 沒有特判空字串，回傳 [""]
  • 忘記 7 和 9 有 4 個字母（pqrs、wxyz），只寫 3 個
  • 忘記撤銷選擇
  • Java 用 String 拼接而非 StringBuilder，效能較差

回溯題型的三大結構

📝 認出結構就知道怎麼寫
【型 A：從同一個集合中選】
  for (i = start 或 0; i < n; i++)
  代表題：#46 排列、#78 子集、#39 組合總和
  差別只在起點與 used 陣列

【型 B：每層有不同的選擇列表】← 本題
  for (choice : getChoices(depth))
  代表題：#17 電話號碼、#79 單詞搜尋（四個方向）

【型 C：分割問題（決定「切在哪」）】
  for (end = start; end < n; end++) 取 s[start..end]
  代表題：#131 回文分割、#93 復原 IP 位址

#93 Restore IP Addresses（分割型範例）

void backtrack(string& s, int start, vector<string>& segments) {
    if (segments.size() == 4) {
        if (start == s.size()) result.push_back(join(segments, "."));
        return;                       // 剛好 4 段且用完所有字元才合法
    }

    // 每段長度 1~3
    for (int len = 1; len <= 3 && start + len <= s.size(); len++) {
        string seg = s.substr(start, len);
        if (seg.size() > 1 && seg[0] == '0') break;   // 不能有前置零
        if (stoi(seg) > 255) break;                    // 不能超過 255

        segments.push_back(seg);
        backtrack(s, start + len, segments);
        segments.pop_back();
    }
}

剪枝條件（前置零、>255、段數）是這題的重點。

實務應用：T9 輸入法

早期功能手機的 T9 預測輸入就是這題的反向應用：使用者按下 4663，系統要在所有 4×3×3×3 = 108 種字母組合中，快速找出哪些是真正的英文單字。實務做法是預先把字典中每個單字轉成數字序列建成雜湊表，查詢時 O(1) 直接取出候選單字，而不是列舉所有組合再逐一查字典 —— 又是一次「預處理換查詢速度」的體現。
