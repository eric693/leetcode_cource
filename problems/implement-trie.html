#208 Implement Trie (Prefix Tree)

Medium
Design Trie Hash Table String

題目描述

Trie（前綴樹 / 字典樹）是一種樹狀資料結構，用於高效地儲存和檢索字串資料集中的鍵。這一資料結構有相當多的應用情境，例如自動補全和拼字檢查。

請你實現 Trie 類別：

  • Trie()：初始化前綴樹物件
  • insert(word)：向前綴樹中插入字串 word
  • search(word)：如果 word 在前綴樹中，返回 true
  • startsWith(prefix)：如果之前已插入的字串中有以 prefix 為前綴的，返回 true

範例

範例：
Trie trie = new Trie();
trie.insert("apple");
trie.search("apple");     // 返回 true
trie.search("app");       // 返回 false（app 只是前綴，不是完整單字）
trie.startsWith("app");   // 返回 true
trie.insert("app");
trie.search("app");       // 返回 true（現在 app 也是完整單字了）

解題思路

💡 核心：把共同前綴合併成同一條路徑
插入 "apple", "app", "apt", "bat" 後的樹：

                root
               /    \
              a      b
              |      |
              p      a
             / \     |
            p   t*   t*
            |
            l  ← 這裡 isEnd = true（"app"）
            |
            e*

* 表示 isEnd = true（該節點是某個單字的結尾）

關鍵設計：
1. 邊代表字元，節點本身不存字元
2. 每個節點有 26 個子節點指標（小寫英文字母）
3. isEnd 標記：區分「完整單字」與「只是前綴」

三個操作的差異只在最後一步

🎯 共用一個 find 函數
insert(word)：
    沿著字元往下走，節點不存在就「建立」
    走完後把最後一個節點的 isEnd 設為 true

search(word)：
    沿著字元往下走，中途斷掉就返回 false
    走完後檢查 node.isEnd 是否為 true

startsWith(prefix)：
    沿著字元往下走，中途斷掉就返回 false
    走完後直接返回 true（不管 isEnd）

→ search 和 startsWith 只差在「要不要檢查 isEnd」

為什麼不用 HashSet 就好？

🔍 Trie 的價值在「前綴查詢」
操作            HashSet          Trie
--------------------------------------------------
精確查詢        O(L) 平均        O(L) 最壞
前綴查詢        O(n×L) 掃全部    O(L)  ★
列出某前綴的所有單字  不支援      O(L + 結果數) ★
按字典序遍歷    需排序 O(n log n) 直接 DFS O(總字元數)
空間            存完整字串       共享前綴，可能更省

L = 字串長度，n = 單字數量
Trie 的最壞查詢是 O(L)，不像 HashSet 有雜湊碰撞的風險。

程式碼實現

class Trie:

    def __init__(self):
        self.children = {}      # 字元 -> Trie 節點
        self.is_end = False     # 是否為某個完整單字的結尾

    def insert(self, word: str) -> None:
        node = self
        for ch in word:
            if ch not in node.children:
                node.children[ch] = Trie()      # 節點不存在就建立
            node = node.children[ch]
        node.is_end = True

    def search(self, word: str) -> bool:
        node = self._find(word)
        return node is not None and node.is_end   # 必須是完整單字

    def startsWith(self, prefix: str) -> bool:
        return self._find(prefix) is not None     # 存在路徑即可

    def _find(self, s: str):
        # 沿著 s 往下走，回傳終點節點；走不通回傳 None
        node = self
        for ch in s:
            if ch not in node.children:
                return None
            node = node.children[ch]
        return node
class Trie {
private:
    Trie* children[26];    // 只處理小寫 a-z
    bool isEnd;

    // 沿著 s 往下走，回傳終點節點；走不通回傳 nullptr
    Trie* find(const string& s) {
        Trie* node = this;
        for (char c : s) {
            int idx = c - 'a';
            if (node->children[idx] == nullptr) return nullptr;
            node = node->children[idx];
        }
        return node;
    }

public:
    Trie() : isEnd(false) {
        for (int i = 0; i < 26; i++) children[i] = nullptr;
    }

    void insert(string word) {
        Trie* node = this;
        for (char c : word) {
            int idx = c - 'a';
            if (node->children[idx] == nullptr) {
                node->children[idx] = new Trie();   // 節點不存在就建立
            }
            node = node->children[idx];
        }
        node->isEnd = true;
    }

    bool search(string word) {
        Trie* node = find(word);
        return node != nullptr && node->isEnd;      // 必須是完整單字
    }

    bool startsWith(string prefix) {
        return find(prefix) != nullptr;             // 存在路徑即可
    }
};
class Trie {
    private Trie[] children;   // 26 個小寫字母
    private boolean isEnd;

    public Trie() {
        children = new Trie[26];
        isEnd = false;
    }

    public void insert(String word) {
        Trie node = this;
        for (char c : word.toCharArray()) {
            int idx = c - 'a';
            if (node.children[idx] == null) {
                node.children[idx] = new Trie();    // 節點不存在就建立
            }
            node = node.children[idx];
        }
        node.isEnd = true;
    }

    public boolean search(String word) {
        Trie node = find(word);
        return node != null && node.isEnd;          // 必須是完整單字
    }

    public boolean startsWith(String prefix) {
        return find(prefix) != null;                // 存在路徑即可
    }

    // 沿著 s 往下走，回傳終點節點；走不通回傳 null
    private Trie find(String s) {
        Trie node = this;
        for (char c : s.toCharArray()) {
            int idx = c - 'a';
            if (node.children[idx] == null) return null;
            node = node.children[idx];
        }
        return node;
    }
}
// 字典樹：每個節點有 26 個子節點與結尾標記
// insert / search / startsWith 皆為 O(L)，空間複雜度: O(總字元數 × 26)
class TrieNode {
    children: (TrieNode | null)[] = new Array(26).fill(null);
    isEnd = false;
}

class Trie {
    private root = new TrieNode();

    insert(word: string): void {
        let node = this.root;
        for (const ch of word) {
            const i = ch.charCodeAt(0) - 97;
            if (!node.children[i]) node.children[i] = new TrieNode();
            node = node.children[i]!;
        }
        node.isEnd = true;
    }

    search(word: string): boolean {
        const node = this.find(word);
        return node !== null && node.isEnd;
    }

    startsWith(prefix: string): boolean {
        return this.find(prefix) !== null;
    }

    private find(s: string): TrieNode | null {
        let node: TrieNode | null = this.root;
        for (const ch of s) {
            node = node.children[ch.charCodeAt(0) - 97];
            if (!node) return null;
        }
        return node;
    }
}
// 字典樹：每個節點有 26 個子節點與結尾標記
// insert / search / startsWith 皆為 O(L)，空間複雜度: O(總字元數 × 26)
typedef struct Trie {
    struct Trie* children[26];
    bool isEnd;
} Trie;

Trie* trieCreate() {
    return calloc(1, sizeof(Trie));          // calloc 會把子節點全設為 NULL
}

void trieInsert(Trie* obj, char* word) {
    Trie* node = obj;
    for (char* p = word; *p; p++) {
        int i = *p - 'a';
        if (!node->children[i]) node->children[i] = trieCreate();
        node = node->children[i];
    }
    node->isEnd = true;
}

static Trie* find(Trie* obj, char* s) {
    Trie* node = obj;
    for (char* p = s; *p && node; p++) node = node->children[*p - 'a'];
    return node;
}

bool trieSearch(Trie* obj, char* word) {
    Trie* node = find(obj, word);
    return node && node->isEnd;
}

bool trieStartsWith(Trie* obj, char* prefix) {
    return find(obj, prefix) != NULL;
}

void trieFree(Trie* obj) {
    if (!obj) return;
    for (int i = 0; i < 26; i++) trieFree(obj->children[i]);
    free(obj);
}

複雜度分析

時間複雜度：
  • insert：O(L)，L 為字串長度
  • search：O(L)
  • startsWith：O(L)
  • 全部與字典中的單字數量 n 無關，這是 Trie 最大的優勢
空間複雜度：O(總字元數 × 字元集大小)
  • 陣列版：每個節點固定 26 個指標，可能浪費
  • HashMap 版：只存實際用到的分支，較省空間但常數較大

重點提示

💡 關鍵技巧 1：isEnd 標記不可省

沒有 isEnd 就無法區分 search("app") 和 startsWith("app")。這是本題的核心設計，也是最常被漏掉的地方。

💡 關鍵技巧 2：陣列 vs HashMap

字元集小且固定（如 a-z）用陣列，查詢快；字元集大或稀疏（Unicode、單字很少）用 HashMap，省空間。面試時可以主動比較這兩種選擇。

💡 面試延伸：如何實作 delete？

先走到結尾把 isEnd 設為 false；再從下往上，若某節點沒有子節點且 isEnd 為 false，就可以刪除它。要小心不能刪掉其他單字還在用的節點。

⚠️ 常見錯誤：
  • search 忘記檢查 isEnd，變成和 startsWith 一樣
  • C++ 忘記在建構子把 children 初始化為 nullptr（未初始化指標是災難）
  • C++ 沒有寫解構子釋放子節點造成記憶體洩漏（LeetCode 不會抓，但面試會被問）
  • 誤以為根節點要存字元；根節點是空的，字元存在「邊」上

補充說明

延伸一：二進位 Trie（0/1 Trie）

🔢 把數字當成 32 位元字串
每個節點只有 2 個子節點（0 和 1），從最高位往下插入。

應用：求最大 XOR 對（#421）
  查詢時貪心地往「與當前位相反」的分支走，
  因為 XOR 要讓高位盡量是 1。
  複雜度從暴力 O(n²) 降到 O(32n)。

延伸二：實務應用

  • 搜尋引擎自動補全：輸入前綴，DFS 收集該子樹下所有單字
  • IP 路由表（Routing Trie）：路由器用二進位 Trie 做最長前綴匹配（LPM）
  • 輸入法 / 拼字檢查：結合編輯距離做模糊比對
  • 敏感詞過濾：AC 自動機（Aho-Corasick）就是 Trie + KMP 失配指標

Trie 的變形：壓縮字典樹（Radix Tree）

當很多節點只有單一子節點時（例如只存了 "internationalization" 一個字），普通 Trie 會產生一長串無分支的節點。Radix Tree 把這些鏈壓縮成一條邊，大幅節省空間，Linux 核心的 page cache 索引就使用這個結構。
